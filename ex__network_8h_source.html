51int send_net_datas(NETWORK* netw, N_STR* data) {
52 __n_assert(netw, return FALSE);
53 __n_assert(data, return FALSE);
54
55 NETW_MSG* msg = NULL;
56 N_STR* tmpstr = NULL;
57 N_STR* hostname = NULL;
58
59 create_msg(&msg);
60 __n_assert(msg, return FALSE);
61
62 hostname = new_nstr(1024);
63#ifdef __windows__
64 if (gethostname(hostname->data, (int)hostname->length) != 0) {
65#else
66 if (gethostname(hostname->data, hostname->length) != 0) {
67#endif
68 n_log(LOG_ERR, "Couldn't resolve hostname. Error:%s", strerror(errno));
69 free_nstr(&hostname);
70 return FALSE;
71 }
72 hostname->written = strlen(hostname->data);
73
74 add_int_to_msg(msg, NETMSG_DATA);
75
76 add_nstrptr_to_msg(msg, hostname);
77
78 n_log(LOG_DEBUG, "Adding string %s", data->data);
79
80 add_nstrdup_to_msg(msg, data);
81
82 for (int i = 0; i < 10; i++) {
83 int val = (rand() % 20) - 10;
84 add_int_to_msg(msg, val);
85 n_log(LOG_DEBUG, "adding %d to message", val);
86 }
87
88 tmpstr = make_str_from_msg(msg);
89
90 delete_msg(&msg);
91
92 __n_assert(tmpstr, return FALSE);
93
94 return netw_add_msg(netw, tmpstr);
95} /* send_net_datas */

104int get_net_datas(N_STR* str, N_STR** hostname, N_STR** data) {
105 NETW_MSG* netmsg = NULL;
106 int type = 0;
107
108 __n_assert(str, return FALSE);
109
110 netmsg = make_msg_from_str(str);
111 __n_assert(netmsg, return FALSE);
112
113 get_int_from_msg(netmsg, &type);
114 if (type != NETMSG_DATA) {
115 n_log(LOG_ERR, "Error: message is not NETMSG_DATA(%d) but %d !", NETMSG_DATA, type);
116 delete_msg(&netmsg);
117 return FALSE;
118 }
119 get_nstr_from_msg(netmsg, &(*hostname));
120 get_nstr_from_msg(netmsg, &(*data));
121 n_log(LOG_DEBUG, "getting string %s", (*data)->data);
122
123 for (int i = 0; i < 10; i++) {
124 int val = 0;
125 get_int_from_msg(netmsg, &val);
126 n_log(LOG_DEBUG, "getting %d from message", val);
127 }
128
129 delete_msg(&netmsg);
130
131 return TRUE;
132} /* get_net_datas( ... ) */

139void* manage_client(void* ptr) {
140 NETWORK* netw = (NETWORK*)ptr;
141 N_STR* netw_exchange = NULL;
142 uint32_t state = 0;
143 int thr_engine_state = 0;
144
145 n_log(LOG_NOTICE, "manage_client started for netw %d", netw->link.sock);
146 netw_start_thr_engine(netw);
147
148 int DONE = 0;
149 while (!DONE) {
150 if ((netw_exchange = netw_get_msg(netw))) {
151 N_STR *hostname = NULL, *data = NULL;
152
153 int type = netw_msg_get_type(netw_exchange);
154 if (type == -1) {
155 n_log(LOG_ERR, "Error getting message type from network exchange");
156 free_nstr(&netw_exchange);
157 continue;
158 }
159 switch (type) {
160 case NETMSG_DATA:
161 get_net_datas(netw_exchange, &hostname, &data);
162 if (hostname && hostname->data && data && data->data) {
163 n_log(LOG_NOTICE, "RECV: %s: %s , %s", netw->link.ip, hostname->data, data->data);
164 send_net_datas(netw, data);
165 } else {
166 n_log(LOG_ERR, "Error decoding request");
167 }
168 break;
169 default:
170 n_log(LOG_ERR, "Unknow message type %d", type);
171 DONE = 1;
172 break;
173 }
174 if (data)
175 free_nstr(&data);
176 if (hostname)
177 free_nstr(&hostname);
178 if (netw_exchange)
179 free_nstr(&netw_exchange);
180 } else {
181 u_sleep(500);
182 }
183 netw_get_state(netw, &state, &thr_engine_state);
184 if ((state & NETW_EXITED) || (state & NETW_ERROR) || (state & NETW_EXIT_ASKED))
185 DONE = 1;
186 } /* while( !DONE ) */
187
188 SOCKET sockid = netw->link.sock;
189 netw_close(&netw);
190 n_log(LOG_NOTICE, "network closed for netw %d !", sockid);
191
192 return NULL;
193} /* manage_client(...) */
